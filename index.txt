        <h2>🎮 Mini-juego: Adivina el Número</h2>
        <p>He pensado un número del 1 al 10. ¿Puedes adivinarlo?</p>
        <input type="number" id="guess" min="1" max="10" placeholder="1-10">
        <br>
        <button onclick="checkGuess()">Probar</button>
        <button class="btn-reset" onclick="resetGame()">Nuevo número</button>
        <p id="resultado"></p>
    </div>

    <script>
        let target = Math.floor(Math.random() * 10) + 1;

        function checkGuess() {
            const val = document.getElementById('guess').value;
            const res = document.getElementById('resultado');
            if (!val) { res.textContent = "⚠️ Escribe un número."; res.style.color = "#e17055"; return; }
            const n = parseInt(val);
            if (n === target) { res.textContent = "🎉 ¡Correcto! ¡Lo lograste!"; res.style.color = "#00b894"; }
            else if (n < target) { res.textContent = "📈 Es más alto..."; res.style.color = "#0984e3"; }
            else { res.textContent = "📉 Es más bajo..."; res.style.color = "#d63031"; }
        }

        function resetGame() {
            target = Math.floor(Math.random() * 10) + 1;
            document.getElementById('guess').value = '';
            document.getElementById('resultado').textContent = '🔄 ¡Nuevo número! Intenta de nuevo.';
            document.getElementById('resultado').style.color = '#636e72';
        }
    </script>
